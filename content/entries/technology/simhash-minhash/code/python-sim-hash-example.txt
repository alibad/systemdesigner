from __future__ import annotations

import hashlib
from collections import Counter
from collections.abc import Mapping


def stable_bits(feature: str, width: int) -> int:
    """Return a deterministic feature hash with the requested bit width."""
    digest_bytes = (width + 7) // 8
    digest = hashlib.blake2b(
        feature.encode("utf-8"),
        digest_size=digest_bytes,
        person=b"simhash-v1",
    ).digest()
    return int.from_bytes(digest, "big") & ((1 << width) - 1)


def weighted_simhash(features: Mapping[str, float], width: int = 64) -> int:
    if width <= 0 or width > 512:
        raise ValueError("width must be between 1 and 512")
    if not features:
        raise ValueError("at least one weighted feature is required")

    accumulators = [0.0] * width
    for feature, weight in features.items():
        if weight < 0:
            raise ValueError("feature weights must be non-negative")
        hashed = stable_bits(feature, width)
        for bit in range(width):
            accumulators[bit] += weight if hashed & (1 << bit) else -weight

    fingerprint = 0
    for bit, score in enumerate(accumulators):
        if score >= 0:
            fingerprint |= 1 << bit
    return fingerprint


def token_counts(text: str) -> Counter[str]:
    # Replace this with the application's versioned normalization contract.
    return Counter(text.casefold().split())


def hamming_distance(left: int, right: int) -> int:
    return (left ^ right).bit_count()


left = weighted_simhash(token_counts("the quick brown fox jumps"))
right = weighted_simhash(token_counts("the quick brown fox leaps"))
distance = hamming_distance(left, right)
print({"distance": distance, "within_radius_3": distance <= 3})
