FOR account IN accounts
  FILTER account.status == "active"
  FOR owner, ownership, path IN 1..2 OUTBOUND account
    GRAPH "business_ownership"
    FILTER owner.jurisdiction == @jurisdiction
  RETURN {
    account: KEEP(account, "_key", "legal_name"),
    owner: KEEP(owner, "_key", "name", "jurisdiction"),
    ownership_percent: ownership.percent,
    evidence_path: path.edges[*]._id
  }
